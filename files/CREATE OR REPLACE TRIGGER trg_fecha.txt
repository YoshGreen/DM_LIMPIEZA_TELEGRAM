CREATE OR REPLACE TRIGGER trg_delete_materia
BEFORE DELETE ON MATERIA
FOR EACH ROW
DECLARE
    v_total NUMBER;
BEGIN
    SELECT COUNT(*)
    INTO v_total
    FROM INSCRIPCION
    WHERE materia_id=:OLD.materia_id;

    IF v_total>0 THEN
        RAISE_APPLICATION_ERROR(-20002,'La materia tiene inscripciones.');
    END IF;
END;
/

CREATE OR REPLACE TRIGGER trg_correo_unico
BEFORE INSERT OR UPDATE ON DOCENTE
FOR EACH ROW
DECLARE
v_total NUMBER;
BEGIN
SELECT COUNT(*)
INTO v_total
FROM DOCENTE
WHERE correo=:NEW.correo
AND docente_id<>NVL(:NEW.docente_id,0);

IF v_total>0 THEN
RAISE_APPLICATION_ERROR(-20012,'Correo duplicado.');
END IF;
END;
/

INTO v_total
FROM INSCRIPCION

END;
